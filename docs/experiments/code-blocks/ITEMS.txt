148|export type PublishResult =
149|  | { status: "PUBLISHED" }
150|  | { status: "ALREADY_PUBLISHED" }
151|  | { status: "COLLISION_REFUSED" }
152|  | { status: "RETRY_EXHAUSTED" };
154|export async function publishMessage(
155|  payloadBytes: Buffer,
156|  targetName: string,
157|  relayRoot: string,
158|  maxRetries = 3
159|): Promise<PublishResult> {
160|  const inDir = join(relayRoot, "in");
161|  const tmpDir = join(relayRoot, "tmp");
162|  const targetPath = join(inDir, targetName);
164|  await fs.mkdir(tmpDir, { recursive: true });
165|  await fs.mkdir(inDir, { recursive: true });
167|  for (let attempt = 0; attempt < maxRetries; attempt++) {
168|    const tmpPath = join(
169|      tmpDir,
170|      `.dep-${process.pid}-${Date.now()}-${Math.random().toString(36).slice(2)}`
171|    );
173|    let tmpCreated = false;
174|    let tmpHandle: fs.FileHandle | null = null;
175|    let dirHandle: fs.FileHandle | null = null;
177|    try {
178|      tmpHandle = await fs.open(tmpPath, "wx");
179|      tmpCreated = true;
180|      await tmpHandle.writeFile(payloadBytes);
181|      await tmpHandle.sync();
182|      await tmpHandle.close();
183|      tmpHandle = null;
185|      try {
186|        await fs.link(tmpPath, targetPath);
187|      } catch (linkErr: any) {
188|        if (linkErr.code === "EEXIST") {
189|          let existingBytes: Buffer;
190|          try {
191|            existingBytes = await fs.readFile(targetPath);
192|          } catch (readErr: any) {
193|            if (readErr.code === "ENOENT") continue;   // target vanished; the name is free
194|            throw readErr;
195|          }
197|          const existingDigest = createHash("sha256").update(existingBytes).digest("hex");
198|          const payloadDigest = createHash("sha256").update(payloadBytes).digest("hex");
200|          if (existingDigest === payloadDigest) {
201|            dirHandle = await fs.open(inDir, "r");
202|            await dirHandle.sync();
203|            return { status: "ALREADY_PUBLISHED" };
204|          }
205|          return { status: "COLLISION_REFUSED" };
206|        }
207|        throw linkErr;
208|      }
210|      dirHandle = await fs.open(inDir, "r");
211|      await dirHandle.sync();
212|      return { status: "PUBLISHED" };
213|    } finally {
214|      if (tmpHandle) await tmpHandle.close().catch(() => {});
215|      if (dirHandle) await dirHandle.close().catch(() => {});
216|      if (tmpCreated) await fs.unlink(tmpPath).catch(() => {});
217|    }
218|  }
220|  return { status: "RETRY_EXHAUSTED" };

297|export type CausalStatus =
298|  | "NO_PARENT" | "UNANCHORED" | "LABEL_ONLY"
299|  | "MATCHES"   | "DIVERGES"   | "UNCHECKABLE";
301|export function evaluateCausalLink(
302|  child: RelayAct,
303|  localStore: Map<string, StoredRecord>
304|): CausalStatus {
305|  const { parent_id, parent_digest } = child;
307|  if (parent_id === null) {
308|    if (parent_digest === null) return "NO_PARENT";
309|    return "UNANCHORED";
310|  }
311|  if (parent_digest === null) return "LABEL_ONLY";
313|  const parentRecord = localStore.get(parent_id);
314|  if (parentRecord !== undefined) {
315|    return parentRecord.digest === parent_digest ? "MATCHES" : "DIVERGES";
316|  }
317|  return "UNCHECKABLE";
